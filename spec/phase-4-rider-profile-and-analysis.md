# Phase 4 — Rider Profile + Goal-Aware Coach + Daily Readiness

> This document is written so another LLM-coding agent (Claude Code, Cursor)
> can pick up any sub-phase and implement it without further context.
> Follow the numbered sub-phases in order; each is independently testable
> before starting the next.

---

## Context & Goal

Phases 1–3 built an ingest pipeline, a per-ride AI analysis, and a React UI.
The coach currently treats the rider as an anonymous data set — it knows
nothing about their preferred discipline, home training terrain, or goals.
Phase 4 adds a **rider-profile layer** that is stored persistently and injected
into every coach interaction, making suggestions concretely relevant to what
the rider is trying to achieve.

Three additions:

1. **Rider profile** — a single DB row capturing discipline (road/mtb/gravel),
   home city, multi-select goals, and free-text notes.
2. **Goal-aware analysis** — profile text prepended to every LLM interaction so
   "you rode a Z2 endurance block" becomes "good Z2 block for your climbing-
   endurance goal; here's how it moves the needle."
3. **Daily 3pm readiness summary** — a short daily note generated by Soft Floyd
   even when no ride happened, covering HRV trend, load, and a goal-specific
   training recommendation. Plus a manual "Sync Garmin now" button so the rider
   doesn't wait 10 min after uploading a ride.

**Single user, SQLite, macOS, local — all Phase 1–3 constraints apply.**

## Global Decisions

| Area | Choice |
|---|---|
| Profile storage | Single-row SQLite table `rider_profile`, guarded by `CHECK (id=1)` |
| Goals format in DB | `goals_json TEXT` — JSON array of slugs from `GOAL_SLUGS` |
| Profile in prompts | User message (not system prompt) so OpenAI auto-caching still hits the static system block |
| City geocoding | Free-text only; no external API |
| Daily summary storage | Separate `daily_summary` table; NOT reusing `Conversation`/`Message` |
| Concurrency guard | `asyncio.Lock` shared between manual sync and the 10-min poller |
| Daily scheduler | New file `src/coach/ingest/daily_scheduler.py` using `apscheduler` `AsyncIOScheduler`; mounted alongside the poller in `coach run` |

```python
"""Profile and daily-summary FastAPI routes."""
from __future__ import annotations

import datetime
import json
from typing import Annotated, Literal

from fastapi import APIRouter, Depends, HTTPException, Query
from pydantic import BaseModel
from sqlalchemy import select
from sqlalchemy.orm import Session

from coach.store.models import DailySummary, RiderProfile
from coach.store.session import get_sync_session

GOAL_SLUGS = Literal[
    "climbing", "descending", "endurance", "sprinting", "intervals", "recovery"
]

router = APIRouter()


class RiderProfileIn(BaseModel):
    discipline: Literal["road", "mtb", "gravel"]
    city: str | None = None
    country: str | None = None
    terrain_notes: str | None = None
    goals: list[GOAL_SLUGS] = []
    freeform_notes: str | None = None


@router.get("/api/profile")
def get_profile():
    session = get_sync_session()
    row = session.get(RiderProfile, 1)
    if row is None:
        raise HTTPException(status_code=404, detail="Profile not set")
    return {
        "discipline": row.discipline,
        "city": row.city,
        "country": row.country,
        "terrain_notes": row.terrain_notes,
        "goals": row.goals,
        "freeform_notes": row.freeform_notes,
        "updated_at": row.updated_at.isoformat(),
    }


@router.put("/api/profile")
def put_profile(body: RiderProfileIn):
    session = get_sync_session()
    row = session.get(RiderProfile, 1)
    if row is None:
        row = RiderProfile(id=1)
        session.add(row)
    row.discipline = body.discipline
    row.city = body.city
    row.country = body.country
    row.terrain_notes = body.terrain_notes
    row.goals = list(body.goals)
    row.freeform_notes = body.freeform_notes
    row.updated_at = datetime.datetime.now(datetime.UTC)
    session.commit()
    return {"ok": True}


@router.get("/api/summary/daily")
def get_daily_summary(date: str | None = Query(None)):
    session = get_sync_session()
    target = datetime.date.fromisoformat(date) if date else datetime.date.today()
    row = session.execute(
        select(DailySummary).where(DailySummary.date == target)
    ).scalar_one_or_none()
    if row is None:
        raise HTTPException(status_code=404, detail="No summary for that date")
    return {
        "date": str(row.date),
        "text": row.text,
        "generated_at": row.generated_at.isoformat(),
        "cost_usd": row.cost_usd,
    }


@router.post("/api/summary/daily")
async def post_daily_summary(date: str | None = Query(None)):
    """Force-generate a daily summary (useful before the scheduler exists)."""
    from coach.agent.daily import generate_daily_summary
    from coach.config import get_config
    from coach.store.session import get_sync_session

    cfg = get_config()
    if not cfg.openai_api_key:
        raise HTTPException(status_code=503, detail="openai_api_key not set")
    session = get_sync_session()
    target = datetime.date.fromisoformat(date) if date else datetime.date.today()
    summary = await generate_daily_summary(session, cfg, target)
    return {"date": str(summary.date), "text": summary.text}
```

```python
"""Build a concise rider-profile string for prompt injection."""
from __future__ import annotations

from sqlalchemy.orm import Session

from coach.store.models import RiderProfile


def build_profile_summary(session: Session) -> str | None:
    row = session.get(RiderProfile, 1)
    if row is None:
        return None
    parts = [f"Discipline: {row.discipline}."]
    if row.city:
        loc = row.city
        if row.country:
            loc += f", {row.country}"
        parts.append(f"Trains in {loc}.")
    if row.terrain_notes:
        parts.append(f"Terrain: {row.terrain_notes}.")
    if row.goals:
        parts.append(f"Goals: {', '.join(row.goals)}.")
    if row.freeform_notes:
        parts.append(f"Notes: {row.freeform_notes}.")
    return " ".join(parts)
```

# SUB-PHASE 4.5 — Daily 3pm readiness summary

### New prompt: `src/coach/agent/prompts/daily_system.md`

```markdown
You are Soft Floyd, a kind and encouraging personal cycling coach. Your task is
to write a **daily readiness note** — not a ride analysis — in exactly two short
paragraphs (3–5 sentences each).

Paragraph 1 — Recovery status: interpret HRV vs. baseline, sleep quality, RHR,
Body Battery, and recent load (ACWR). Use the same tone as the main persona:
honest, never alarming, celebrates good recovery.

Paragraph 2 — Today's recommendation: based on the rider's goals and recovery
```

### New file: `src/coach/agent/daily.py`

```python
"""Daily readiness summary generation for Soft Floyd."""
from __future__ import annotations

import datetime
from pathlib import Path

import openai

from coach.agent.profile import build_profile_summary
from coach.log import log
from coach.store.models import Activity, DailySummary, WellnessDaily
from coach.web.cost import calculate_cost, record_usage

_DAILY_SYSTEM_MD = Path(__file__).parent / "prompts" / "daily_system.md"
_MODEL = "gpt-4.1-mini"
_MAX_TOKENS = 350


def build_daily_context(session, cfg, date: datetime.date) -> str:
    """Compose the user message for the daily summary call."""
    from sqlalchemy import select
    import datetime as dt

    # Wellness for today
    wellness_row = session.execute(
        select(WellnessDaily).where(WellnessDaily.date == date)
    ).scalar_one_or_none()

    # Last 7 days of activity load
    cutoff = datetime.datetime.combine(date - dt.timedelta(days=7), dt.time.min, dt.UTC)
    activities = list(
        session.execute(
            select(Activity)
            .where(Activity.start_time >= cutoff)
            .order_by(Activity.start_time.desc())
            .limit(7)
        ).scalars()
    )

    profile_text = build_profile_summary(session) or "(no profile set)"

    lines = [f"## Date: {date}", f"## Rider Profile\n{profile_text}"]

    if wellness_row:
        lines.append(
            f"## Today's Wellness\n"
            f"HRV overnight: {wellness_row.hrv_overnight}, "
            f"Sleep score: {wellness_row.sleep_score}, "
            f"RHR: {wellness_row.resting_hr}, "
            f"Body Battery: {wellness_row.body_battery_low}→{wellness_row.body_battery_high}, "
            f"ACWR: {wellness_row.acwr}"
        )
    else:
        lines.append("## Today's Wellness\n(no data yet)")

    if activities:
        load_lines = [f"- {a.start_time.date()} {a.bike_type} {round(a.duration_s/60)}min tss={a.tss_proxy or '?'}"
                      for a in activities]
        lines.append("## Recent Load (last 7 days)\n" + "\n".join(load_lines))

    return "\n\n".join(lines)


async def generate_daily_summary(session, cfg, date: datetime.date) -> DailySummary:
    """Generate (or regenerate) the daily summary for a given date."""
    client = openai.AsyncOpenAI(api_key=cfg.openai_api_key)
    system = _DAILY_SYSTEM_MD.read_text(encoding="utf-8")
    user_content = build_daily_context(session, cfg, date)

    response = await client.chat.completions.create(
        model=_MODEL,
        max_tokens=_MAX_TOKENS,
        messages=[
            {"role": "system", "content": system},
            {"role": "user", "content": user_content},
        ],
        stream=False,
    )

    text = (response.choices[0].message.content or "").strip()
    usage = response.usage

    prompt_tokens = getattr(usage, "prompt_tokens", 0) or 0
    tokens_out = getattr(usage, "completion_tokens", 0) or 0
    details = getattr(usage, "prompt_tokens_details", None)
    cache_read = getattr(details, "cached_tokens", 0) or 0
    uncached = prompt_tokens - cache_read
    cost = calculate_cost(uncached, tokens_out, cache_read)

    # Upsert row for the date
    from sqlalchemy import select as _select
    row = session.execute(
        _select(DailySummary).where(DailySummary.date == date)
    ).scalar_one_or_none()
    if row is None:
        row = DailySummary(date=date)
        session.add(row)
    row.text = text
    row.tokens_in = prompt_tokens
    row.tokens_out = tokens_out
    row.cache_read = cache_read
    row.cost_usd = float(cost)
    row.generated_at = datetime.datetime.now(datetime.UTC)
    session.commit()

    log.info("daily_summary.generated", date=str(date), cost_usd=float(cost))
    return row
```

### New file: `src/coach/ingest/daily_scheduler.py`

```python
"""APScheduler cron that fires the daily readiness summary at 15:00 local time."""
from __future__ import annotations

import datetime

from apscheduler.schedulers.asyncio import AsyncIOScheduler
from apscheduler.triggers.cron import CronTrigger

from coach.log import log


def start_daily_scheduler(cfg) -> AsyncIOScheduler:
    scheduler = AsyncIOScheduler()

    async def _run() -> None:
        from coach.agent.daily import generate_daily_summary
        from coach.store.session import get_sync_session

        if not cfg.openai_api_key:
            log.warning("daily_scheduler.skipped_no_key")
            return
        session = get_sync_session()
        today = datetime.date.today()
        try:
            await generate_daily_summary(session, cfg, today)
        except Exception as exc:
            log.error("daily_scheduler.failed", error=str(exc))

    scheduler.add_job(_run, CronTrigger(hour=15, minute=0))
    scheduler.start()
    log.info("daily_scheduler.started", trigger="15:00 local")
    return scheduler
```

# SUB-PHASE 4.6 — React onboarding + settings

**Goal:** On first launch (no profile row), the rider sees an onboarding form.
Settings are always reachable from the nav bar.

### `frontend/src/lib/useProfile.ts`

```ts
import { useState, useEffect } from "react";

export type Profile = {
  discipline: "road" | "mtb" | "gravel";
  city?: string;
  country?: string;
  terrain_notes?: string;
  goals: string[];
  freeform_notes?: string;
};

export function useProfile() {
  const [profile, setProfile] = useState<Profile | null | undefined>(undefined);

  useEffect(() => {
    fetch("/api/profile")
      .then((r) => (r.ok ? r.json() : null))
      .then(setProfile)
      .catch(() => setProfile(null));
  }, []);

  return { profile, refetch: () => fetch("/api/profile").then(r => r.ok ? r.json() : null).then(setProfile) };
}
```

- `undefined` = loading; `null` = 404 (no profile); `Profile` = loaded.

### Route guard in `frontend/src/App.tsx`

# SUB-PHASE 4.7 — React sync button + daily readiness widget

**Goal:** The activity list shows today's readiness summary and a manual Garmin
sync button.

### `frontend/src/components/DailyReadiness.tsx`

```tsx
import { useState, useEffect } from "react";

export default function DailyReadiness() {
  const [data, setData] = useState<{ text: string; date: string } | null>(null);
  const [regen, setRegen] = useState(false);

  const load = () =>
    fetch("/api/summary/daily")
      .then((r) => (r.ok ? r.json() : null))
      .then(setData);

  useEffect(() => { load(); }, []);

  const regenerate = async () => {
    setRegen(true);
    await fetch("/api/summary/daily", { method: "POST" });
    await load();
    setRegen(false);
  };

  if (!data) return null;
  return (
    <div className="mb-6 rounded-xl border border-blue-100 bg-blue-50 p-4 text-sm text-gray-700">
      <div className="mb-1 flex items-center justify-between">
        <span className="font-medium text-blue-700">Today's readiness</span>
        <button onClick={regenerate} disabled={regen}
          className="text-xs text-blue-400 hover:text-blue-700 disabled:opacity-40">
          {regen ? "Regenerating…" : "Regenerate"}
        </button>
      </div>
      <p className="whitespace-pre-wrap">{data.text}</p>
    </div>
  );
}
```

### Sync button on `frontend/src/pages/ActivityList.tsx`

```tsx
const [syncing, setSyncing] = useState(false);
const [syncMsg, setSyncMsg] = useState<string | null>(null);

const syncGarmin = async () => {
  setSyncing(true);
  setSyncMsg(null);
  try {
    const r = await fetch("/api/sync/garmin", { method: "POST" });
    const data = await r.json();
    if (data.new_activity_ids?.length > 0) {
      navigate(`/activities/${data.new_activity_ids[0]}`);
    } else {
      setSyncMsg("No new activities found.");
    }
  } finally {
    setSyncing(false);
  }
};
```

Add above the activity list:
```tsx
<DailyReadiness />
<div className="mb-4 flex items-center gap-3">
  <button onClick={syncGarmin} disabled={syncing}
    className="rounded bg-indigo-600 px-4 py-2 text-sm text-white hover:bg-indigo-700 disabled:opacity-50">
    {syncing ? "Syncing…" : "Sync Garmin now"}
  </button>
  {syncMsg && <span className="text-sm text-gray-500">{syncMsg}</span>}
</div>
```

### Acceptance

1. `GET /api/summary/daily` returns text → readiness card visible on `/activities`.
2. Click "Sync Garmin now" → spinner, then either routes to new activity or
   shows "No new activities found."
3. Click "Regenerate" → card updates after ~2s.
